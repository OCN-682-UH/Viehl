eDNA Data Cleaning Function:

See below for the details regarding the data cleaning function - clean_eDNA()

Show code 
# set up the function with the following variables:
#    data = your dataframe
#    first_sample_ID = the first sample column name when reading from left to right
#    control_IDs = any negative or positive control column names, as they vary from file to file and thus should be specified in this case. If my data entry organization was better in the past, I wouldn't have to have this!

clean_eDNA <- function(data, first_sample_ID, control_IDs) {
  
  data |> 
    #remove everything that does not have a domain
    filter(domain != "" | domain != "dropped") -> noblank_data
    
  noblank_data |> 
    #pull the names of the sample columns
    select({{first_sample_ID}}:last_col()) |> 
    colnames() -> sample_cols
  
  # Add a new row called neg_totals that is the sum of every negative column specified
  noblank_data |> 
    mutate(neg_totals = rowSums(noblank_data[, {{control_IDs}}], na.rm = TRUE)) |> 

  # Edit all sample columns to subtract this new negative sum column
    mutate(across(all_of(sample_cols), ~ .x - neg_totals)) |> 

  # change all negative numbers across all of the sample columns to zeros
    mutate(across(all_of(sample_cols), ~ ifelse(.x < 0, 0, .x))) |> 
  
  # remove the neg_totals row
    select(1:(last_col()-1)) -> clean_need_sum
  
  # add sum column of all sample columns
  clean_need_sum |> 
    mutate(sample_sums = rowSums(clean_need_sum[, sample_cols], na.rm = TRUE)) |> 
  
  # remove any rows that have a zero sum after subtracting negative controls
    filter(sample_sums != 0) |> 
    
  # remove the sample_sums row
    select(1:(last_col()-1)) |> 
  
  # remove all negative control columns 
    select(-all_of({{control_IDs}})) -> cleaned_eDNA
  
  return(cleaned_eDNA)
}

Plotting Function

For this section, I will be making a function that pivots long and plots the eDNA data cleaned from the previous function for quick visualization. I would like to make it compatible with both the clean and pre-cleaned data, for easy comparisons.

Show code 
# create function
plot_eDNA <- function(data, first_sample_ID, plotby_taxa = phylum, stack_or_fill = "stack",
                      domains = "none", kingdoms = "none", phyla = "none", classes = "none",
                      orders = "none", families = "none", genera = "none", specieses = "none") {
  
  # set up labels for later based on argument inputs
  if (stack_or_fill == "stack") {
    y_axis <- "Number of Sequences"
    title_text <- "eDNA Sequence Summary Plot"
  } else if (stack_or_fill == "fill") {
    y_axis <- "Relative Amount of Sequences"
    title_text <- "eDNA Relative Sequence Summary Plot"
  }
  
  ################ Filter by identities:
  
# if domain is specified, filter the data by the domains specified
  if (length(domains) == 1 && domains == "none") {
    data <- data
  } else {
    data <- filter(data, domain %in% domains)
  }
  
  # if kindom is specified, filter the data by the kingdoms specified
  if (length(kingdoms) == 1 && kingdoms == "none") {
    data <- data
  } else {
    data <- filter(data, kingdom %in% kingdoms)
  }
  
  # if phylum is specified, filter the data by the phyla specified
  if (length(phyla) == 1 && phyla == "none") {
    data <- data
  } else {
    data <- filter(data, phylum %in% phyla)
  }
  
  # if class is specified, filter the data by the classes specified
  if (length(classes) == 1 && classes == "none") {
    data <- data
  } else {
    data <- filter(data, class %in% classes)
  }
  
  # if order is specified, filter the data by the orders specified
  if (length(orders) == 1 && orders == "none") {
    data <- data
  } else {
    data <- filter(data, order %in% orders)
  }
  
  # if family is specified, filter the data by the families specified
  if (length(families) == 1 && families == "none") {
    data <- data
  } else {
    data <- filter(data, family %in% families)
  }
  
  # if genus is specified, filter the data by the genera specified
  if (length(genera) == 1 && genera == "none") {
    data <- data
  } else {
    data <- filter(data, genus %in% genera)
  }
  
  # if species is specified, filter the data by the species specified 
  # hilariously, I had to make this argument "specieses" as I can't have it be "species" D:
  if (length(specieses) == 1 && specieses == "none") {
    data <- data
  } else {
    data <- filter(data, species %in% specieses)
  }
  
  # call data
  data |> 
    
    # retrieve the sample column names again
    select({{first_sample_ID}}:last_col()) |> 
    colnames() -> sample_cols
  
  # call it again
  data |> 
    
    #pivot the data longer to better plot with
    pivot_longer(cols = sample_cols,
                 names_to = "site",
                 values_to = "num_sequences") -> long_eDNA
  
  # and plot!
  long_eDNA |> 
    ggplot(mapping = aes(x = site,
                         y = num_sequences,         # base plot, fill by specified taxa
                         fill = {{plotby_taxa}})) +
    geom_col(position = stack_or_fill) +
    theme_dark() +
    coord_flip() +                               #theming and aethetics
    scale_fill_viridis_d() +
    theme(title = element_text(face = "bold",
                               size = 25),
          axis.title = element_text(size = 15)) +
    labs(title = title_text,
         subtitle = "Use arguments to refine if needed",
         caption = "Note: Any gap between 0 sequences and the column's beginning are unidentified sequences.",
         x = "Site / Sample Name",               # labels
         y = y_axis) -> eDNA_plot
    
  return(eDNA_plot)
  
}
